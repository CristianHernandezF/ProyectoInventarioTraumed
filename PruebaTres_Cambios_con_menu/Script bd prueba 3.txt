DROP DATABASE IF EXISTS bdtraumed;
CREATE DATABASE bdtraumed;
USE bdtraumed;

CREATE TABLE tb_tipo_producto(
    id_tipo INT PRIMARY KEY AUTO_INCREMENT, 
    nombre_tipo VARCHAR(50) NOT NULL 
);


-- Contiene SOLO los datos comunes a todos los productos.
CREATE TABLE tb_producto(
    id_producto INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(100) NOT NULL,
    material VARCHAR(50) NOT NULL,    
    marca VARCHAR(50) NOT NULL,	
    precio INT (6) NOT NULL,
    stock INT (6) NOT NULL,	 	       
    id_tipo INT, 
    
    CONSTRAINT fk_prod_tipo FOREIGN KEY (id_tipo) 
        REFERENCES tb_tipo_producto(id_tipo)
);


CREATE TABLE tb_placa(
    id_placa INT PRIMARY KEY AUTO_INCREMENT,
    id_producto INT NOT NULL UNIQUE,  
    medida DECIMAL(3,1) NOT NULL,    
    cantidad_agujeros INT (6) NOT NULL,
    
    CONSTRAINT fk_placa_prod FOREIGN KEY (id_producto) 
        REFERENCES tb_producto(id_producto) 
        ON DELETE CASCADE             
);


CREATE TABLE tb_tornillo(
    id_tornillo INT PRIMARY KEY AUTO_INCREMENT,
    id_producto INT NOT NULL UNIQUE,  
    diametro DECIMAL(3,1) NOT NULL,  
    largo DECIMAL(3,1) NOT NULL,
    tipo_cabeza VARCHAR(50) NOT NULL, 
    
    CONSTRAINT fk_tornillo_prod FOREIGN KEY (id_producto) 
        REFERENCES tb_producto(id_producto) 
        ON DELETE CASCADE             
);
